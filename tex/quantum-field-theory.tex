\begin{frame}{素材準備: 場の理論}
\footnotesize
\begin{columns}[T]
  \begin{column}{0.51\textwidth}
    {\large\bfseries\color{indigo} クライン-ゴルドン場}
    \par\smallskip
    \textbf{作用} $S$ は 場全体 $\mathcal{F} \coloneqq \{\varphi: \mathbb{R} \to \mathbb{C}\}$ 上の関数
    \begin{minipage}[c]{0.7\columnwidth}
      \begin{center}\hl{S(\varphi) \coloneqq \int_{-\infty}^{\infty} (|\dot{\varphi}|^2 - |\varphi|^2) \, dt}\end{center}
    \end{minipage}\hfill
    \begin{minipage}[c]{0.25\columnwidth}
      \begin{tikzpicture}[node distance=0.4cm]
        \node (target) {$S(\varphi)$};
        \node[draw=blue!50, fill=blue!10, cloud, cloud puffs=11, cloud puff arc=120, minimum width=2cm, minimum height=0.8cm, inner sep=0pt, below=of target] (F) {};
        \node[font=\scriptsize] (phi) at (F.center) {$\bullet\,\varphi$};
        \node[font=\scriptsize, anchor=north west] at ([xshift=3pt]F.south east) {$\mathcal{F}$};
        \draw[|->, thick, indigo] (phi.north) to[bend right=35] (target.south);
      \end{tikzpicture}
    \end{minipage}

    \smallskip
    変分をとると
    \begin{align*}
      \delta S &\color{gray!65} = \int \left[\dot{\varphi}\delta\bar{\dot{\varphi}} + \bar{\dot{\varphi}}\delta\dot{\varphi} - \varphi\delta\bar{\varphi} - \bar{\varphi}\delta\varphi\right] dt \\
      &= -\int \left[(\ddot{\varphi}+\varphi)\delta\bar{\varphi} + \overline{(\ddot{\varphi}+\varphi)}\delta\varphi \right] dt + \underbrace{\Bigl[\underbrace{\dot{\varphi}\delta\bar{\varphi} + \bar{\dot{\varphi}}\delta\varphi}_{=:\,\gamma}\Bigr]_{-\infty}^{\infty}}_{= 0}
    \end{align*}
    \hl{\delta S = 0 \Leftrightarrow \ddot{\varphi} + \varphi = 0} \textbf{クライン-ゴルドン方程式}
  \end{column}
  \begin{column}{0.01\textwidth}
    \rule{0.5pt}{7.5cm}
  \end{column}
  \begin{column}{0.43\textwidth}
    {\large\bfseries\color{indigo} 解空間の構造}
    \par\smallskip
    \begin{center}\hl{\begin{aligned}
      \mathrm{Sol} &\coloneqq \{\varphi \mid \ddot{\varphi}+\varphi=0\}\\
      &= \{pe^{it}+qe^{-it} \mid p,q \in \mathbb{C}\}
    \end{aligned}}\end{center}

    \smallskip
    \begin{center}
    \begin{tikzpicture}[node distance=0.3cm, chipbox/.style={draw=indigo!50, fill=indigo!5, rounded corners, inner sep=3pt}]
      \node[chipbox] (s0) {$e^{it}$};
      \node[chipbox, right=of s0] (s1) {$e^{-it}$};
      \node[draw, rounded corners, fit=(s0)(s1), inner sep=6pt] (souter) {};
      \node[anchor=south, font=\small, inner sep=2pt] at (souter.north) {$\omega_\mathrm{Sol}$};
      \node[anchor=north west, font=\small, inner sep=1pt] at ([xshift=-3pt]souter.south east) {$\mathrm{Sol}$};
    \end{tikzpicture}
    \end{center}

    \smallskip
    $\mathrm{Sol}$ 上の\textbf{シンプレクティック構造}
    \begin{center}\hl{\omega_\mathrm{Sol} \coloneqq 2i(dp \wedge d\bar{p} - dq \wedge d\bar{q})}\end{center}
    は境界項 $\gamma$ の微分から誘導される
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{場の量子論の構成}
  \vspace*{\fill}
  \begin{columns}[T]
    \begin{column}{0.56\textwidth}
      {\large\bfseries\color{indigo} ヒルベルト空間 $\mathcal{H}$}
      \par\bigskip

      解空間 $\mathrm{Sol}$ の \textbf{正エネルギー部分} をとる

      \noindent
      \begin{minipage}[c]{0.7\columnwidth}
      \[ H_+ \coloneqq \{ pe^{it} \mid p \in \mathbb{C} \} \]
      \end{minipage}%
      \hfill
      \begin{minipage}[c]{0.25\columnwidth}
      \vspace{1pt}
      \begin{tikzpicture}[chipbox/.style={draw=indigo!50, fill=indigo!5, rounded corners, inner sep=3pt}]
        \node[chipbox] (h0) {$e^{it}$};
        \node[draw, rounded corners, fit=(h0), inner sep=6pt] (houter) {};
        \node[anchor=south, font=\small, inner sep=2pt] at ([xshift=4pt]houter.north) {$(-, -)_{H_+}$};
        \node[anchor=north west, font=\small, inner sep=1pt] at ([xshift=-3pt]houter.south east) {$H_+$};
      \end{tikzpicture}
      \end{minipage}

      \medskip
      シンプレクティック形式 $\omega_\mathrm{Sol}$ をひねると $H_+$ に \textbf{内積} $(-, -)_{H_+}$ が誘導される

      \bigskip
      \textbf{フォック空間} = \textbf{多項式環}

      \smallskip
      \noindent
      \begin{minipage}[c]{0.515\columnwidth}
      \hl{\mathcal{D} \coloneqq \bigoplus S^n H_+ = \mathbb{C}[e^{it}]}
      \end{minipage}%
      \hspace{6pt}%
      \begin{minipage}[c]{0.4\columnwidth}
      \vspace{8pt}

      $\mathcal{D}$ を完備化して $\mathcal{H} \coloneqq \widehat{\mathcal{D}}$
